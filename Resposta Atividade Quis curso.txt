def realizar_atividade():

    print()
    for indice_p, txtpergunta in perguntas.items():
        print(f'{indice_p}: {txtpergunta["pergunta"]}')

        print('Respostas: ')
        for indice_r, valor_r in txtpergunta['respostas'].items():
            print(f'[{indice_r}]: {valor_r}')

        # Verificando as respostas do usuário
        resposta_usuario = input('Escolha a opção: ').strip()

        if resposta_usuario == txtpergunta['resposta_certa']:
            print('Acertou!\n')
            acertos += 1
        else:
            print(f'Errou! A resposta certa era: {txtpergunta["resposta_certa"]}\n')

    # Resultado final
    print(f'Você acertou {acertos} de {len(perguntas)} perguntas.')
